# ============================================================
# Sparkle Car Wash — how to start everything (Windows PowerShell)
# Repo root: carwash-booking/
# Phase 9: public site + admin + three booking channels
# ============================================================
# Open ONE terminal per section below. Keep them all running.
# Root .env = server secrets (never commit).
# frontend/.env.local = browser-safe NEXT_PUBLIC_* only.
# ============================================================


# ------------------------------------------------------------
# ONE-TIME SETUP (only if new machine / fresh clone)
# ------------------------------------------------------------
# cd backend
# .\venv\Scripts\activate          # or: python -m venv venv
# pip install -r requirements.txt
# alembic upgrade head
# python -m scripts.seed
#
# cd ..\frontend
# copy .env.local.example .env.local
# npm install
#
# cd ..\whatsapp-bridge
# copy .env.example .env           # WHATSAPP_BRIDGE_SECRET must match root .env
# npm install
#
# ngrok: install to C:\ngrok\ngrok.exe and run once:
#   C:\ngrok\ngrok.exe config add-authtoken <your-token>


# ------------------------------------------------------------
# Terminal 1 — Backend (FastAPI)  REQUIRED
# ------------------------------------------------------------
cd backend
.\venv\Scripts\activate
alembic upgrade head
uvicorn app.main:app --reload --host 127.0.0.1 --port 8000

# Health:        http://127.0.0.1:8000/health
# Voice status:  http://127.0.0.1:8000/api/voice/provider
# Admin status:  http://127.0.0.1:8000/api/admin/status
# API docs:      http://127.0.0.1:8000/docs
#
# If port 8000 is stuck (Windows ghost listener):
#   uvicorn app.main:app --reload --host 127.0.0.1 --port 8001
# Then set frontend/.env.local:
#   NEXT_PUBLIC_API_URL=http://localhost:8001
# And point ngrok at 8001:  C:\ngrok\ngrok.exe http 8001


# ------------------------------------------------------------
# Terminal 2 — Frontend (Next.js)  REQUIRED for Phase 9 web app
# ------------------------------------------------------------
cd frontend
# First time / after pull: copy .env.local.example .env.local
# Edit frontend/.env.local:
#   NEXT_PUBLIC_API_URL=http://localhost:8000
#   NEXT_PUBLIC_WHATSAPP_NUMBER=92300...     # digits only, for wa.me
#   NEXT_PUBLIC_VAPI_PUBLIC_KEY=...         # VAPI Dashboard PUBLIC key only
#   NEXT_PUBLIC_VAPI_ASSISTANT_ID=...       # same assistant as root .env
# Never put VAPI_API_KEY / VAPI_WEBHOOK_SECRET / WHATSAPP_BRIDGE_SECRET here.
npm run dev
# Site:   http://localhost:3000
# Book:   http://localhost:3000/book
# Voice:  http://localhost:3000/voice
# Admin:  http://localhost:3000/admin


# ------------------------------------------------------------
# Terminal 3 — WhatsApp bridge (Baileys)  OPTIONAL
# ------------------------------------------------------------
cd whatsapp-bridge
# whatsapp-bridge/.env BACKEND_URL must match uvicorn port (8000 or 8001).
npm start
# Scan QR once. Bridge posts to backend /api/whatsapp/messages
# Needs root .env: WHATSAPP_BRIDGE_SECRET, GEMINI_API_KEY, WHATSAPP_AGENT_MODE=auto


# ------------------------------------------------------------
# Terminal 4 — ngrok (VAPI tool webhooks)  OPTIONAL for live voice
# ------------------------------------------------------------
C:\ngrok\ngrok.exe http 8000
# (Use http 8001 if backend is on 8001.)
# Copy the https://….ngrok-free.dev URL.
# Free URLs often change on restart — update VAPI Server URL each time.


# ------------------------------------------------------------
# VAPI wiring (phone tools + browser voice)
# ------------------------------------------------------------
# Root .env (server — webhook / outbound API):
#   VOICE_PROVIDER=vapi
#   VAPI_API_KEY=...
#   VAPI_ASSISTANT_ID=...
#   VAPI_WEBHOOK_SECRET=...
# Restart Terminal 1 after changing root .env.
#
# VAPI Dashboard → Assistant / Save Booking tool → Server URL:
#   https://<ngrok-host>/api/voice/vapi/webhook
# Auth: Bearer = VAPI_WEBHOOK_SECRET  (NOT VAPI_API_KEY)
#
# Browser /voice page (frontend/.env.local):
#   NEXT_PUBLIC_VAPI_PUBLIC_KEY=<Public key from VAPI dashboard>
#   NEXT_PUBLIC_VAPI_ASSISTANT_ID=<same assistant id>
# Restart Terminal 2 after changing .env.local.
#
# Self-check:
#   GET http://127.0.0.1:8000/api/voice/provider
#   → active_provider=vapi, configured=true
