# ============================================================
# Sparkle Car Wash — how to start everything (Windows PowerShell)
# Repo root: carwash-booking/
# ============================================================
# Open ONE terminal per section below. Keep them all running.
# Root .env holds secrets (never commit it). Copy from .env.example first time.
# ============================================================


# ------------------------------------------------------------
# ONE-TIME SETUP (only if new machine / fresh clone)
# ------------------------------------------------------------
# cd backend
# .\venv\Scripts\activate          # or: python -m venv venv
# pip install -r requirements.txt
# alembic upgrade head
# python -m scripts.seed
#
# cd ..\frontend
# npm install
#
# cd ..\whatsapp-bridge
# copy .env.example .env           # set WHATSAPP_BRIDGE_SECRET to match root .env
# npm install
#
# ngrok: install to C:\ngrok\ngrok.exe and run once:
#   C:\ngrok\ngrok.exe config add-authtoken <your-token>


# ------------------------------------------------------------
# Terminal 1 — Backend (FastAPI)  REQUIRED
# ------------------------------------------------------------
cd backend
.\venv\Scripts\activate
alembic upgrade head
uvicorn app.main:app --reload --port 8000

# Health check: http://127.0.0.1:8000/health
# Voice status:  http://127.0.0.1:8000/api/voice/provider
# If port 8000 is stuck, use --port 8001 and point ngrok at 8001 instead.


# ------------------------------------------------------------
# Terminal 2 — Frontend (Next.js)  OPTIONAL for dashboard/UI
# ------------------------------------------------------------
cd frontend
npm run dev
# Open http://localhost:3000


# ------------------------------------------------------------
# Terminal 3 — WhatsApp bridge (Baileys)  OPTIONAL for WhatsApp
# ------------------------------------------------------------
cd whatsapp-bridge
npm start
# Scan QR once. Bridge posts to backend /api/whatsapp/messages
# Needs root .env: WHATSAPP_BRIDGE_SECRET, GEMINI_API_KEY, WHATSAPP_AGENT_MODE=auto


# ------------------------------------------------------------
# Terminal 4 — ngrok (public URL for VAPI webhooks)  OPTIONAL for voice
# ------------------------------------------------------------
C:\ngrok\ngrok.exe http 8000
# Copy the https://….ngrok-free.app URL from the ngrok UI / terminal.
# Free URLs change when you restart ngrok — update VAPI each time.


# ------------------------------------------------------------
# VAPI voice wiring (after Terminals 1 + 4 are up)
# ------------------------------------------------------------
# In root .env:
#   VOICE_PROVIDER=vapi
#   VAPI_API_KEY=...
#   VAPI_ASSISTANT_ID=...
#   VAPI_WEBHOOK_SECRET=...
# Restart Terminal 1 after changing .env
#
# In VAPI Dashboard → Assistant → Server URL:
#   https://<ngrok-host>/api/voice/vapi/webhook
# Auth: Bearer token = same value as VAPI_WEBHOOK_SECRET
# (Do NOT use VAPI_API_KEY for webhook auth.)
#
# Self-check:
#   GET  http://127.0.0.1:8000/api/voice/provider
#   Should show active_provider=vapi, configured=true


# ------------------------------------------------------------
# Uplift voice (alternative) — set VOICE_PROVIDER=uplift
# ------------------------------------------------------------
# UPLIFT_API_KEY=...
# UPLIFT_AGENT_ID=...
# VOICE_WEBHOOK_SECRET=...
# Tool handlers proxy to: POST /api/voice/uplift/webhook
# See backend/app/voice/README.md
